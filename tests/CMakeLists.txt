cmake_minimum_required(VERSION 3.22)
project(ModbusHostTests C)
enable_testing()
add_executable(modbus_tests modbus_test.c ../HardLib/modbus.c ../HardLib/rs485.c)
target_include_directories(modbus_tests PRIVATE stubs ../HardLib)
set_target_properties(modbus_tests PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(modbus_tests PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME modbus_regressions COMMAND modbus_tests)
